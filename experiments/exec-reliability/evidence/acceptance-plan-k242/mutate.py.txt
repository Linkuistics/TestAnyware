"""Revert one repair at a time in a scratch copy and show that its controls fail."""
import shutil, subprocess, sys
from pathlib import Path

SRC = Path(sys.argv[1]); WORK = Path(sys.argv[2])
MUTATIONS = {
    "R1 single snapshot": ("acceptance_k223_check.py",
        "        failures += child_poll_failures(out)",
        "        failures += [] if child_state(out) == 'absent' else ['known child']"),
    "R2 old inequality": ("acceptance_k223_check.py",
        "    return published_by - (cast(int, duration) + cast(int, finalised)) / 1000",
        "    return cast(float, cancel['ended_after']) - cast(int, duration) / 1000"),
    "R3 no agent check": ("acceptance_k223_check.py",
        "        failures += agent_identity_failures(clone, out)", "        pass"),
    "R3 no app check": ("acceptance_k223_check.py",
        "        return app_identity_failures(clone)", "        return _receipt_failures(clone, name)"),
    "R4 no freeze check": ("acceptance_k223_check.py",
        "    failures += freeze_failures(clone)\n", ""),
    "R5 summed acquire": ("acceptance_k223.py",
        "manifest.tart, manifest.boot),", "manifest.tart, manifest.boot + manifest.slot),"),
    "R6 no payload check": ("acceptance_k223_check.py",
        "        if set(_mapping(leader)) != keys or (", "        if False and ("),
    "R7 no order check": ("acceptance_k223_check.py",
        "    failures += execution_failures(clone)\n", ""),
}
for label, (name, old, new) in MUTATIONS.items():
    if WORK.exists():
        shutil.rmtree(WORK)
    shutil.copytree(SRC, WORK, ignore=shutil.ignore_patterns("evidence", ".venv", "__pycache__"))
    (WORK / "evidence").symlink_to(SRC.resolve() / "evidence")
    path = WORK / name
    text = path.read_text()
    assert text.count(old) == 1, (label, text.count(old))
    path.write_text(text.replace(old, new))
    run = subprocess.run([sys.executable, "-m", "pytest", "-q", "-p", "no:cacheprovider",
                          str(WORK / "test_acceptance_k223.py")], capture_output=True, text=True,
                         cwd=WORK, timeout=300)
    summary = run.stdout.strip().splitlines()[-1]
    print(f"{label}: exit {run.returncode}: {summary}")
